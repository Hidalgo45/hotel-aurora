2 · Diagrama de clases

classDiagram
  direction TB

  class Persona {
    <>
    -_cedula: str
    -_nombres: str
    -_apellidos: str
    -_email: str
    +nombre_completo() str
    +descripcion_rol()* str
    +permisos()* list
  }
  class Usuario {
    -__password_hash: str
    +verificar_clave(clave) bool
    +cambiar_clave(nueva) None
  }
  class Administrador {
    +descripcion_rol() str
    +permisos() list
  }
  class Recepcionista {
    -_turno: str
    +descripcion_rol() str
    +permisos() list
    +registrar_check_in(reserva) None
  }
  class Cliente {
    -_fecha_nacimiento: date
    +descripcion_rol() str
    +permisos() list
    +edad() int
    +es_recurrente() bool
  }

  class Habitacion {
    <>
    -_numero: str
    -_piso: int
    -_tarifa_base: Decimal
    -__estado: EstadoHabitacion
    +estado() EstadoHabitacion
    +cambiar_estado(nuevo) None
    +esta_disponible() bool
    +calcular_tarifa(noches, factor)* Decimal
    +servicios_incluidos()* list
    +capacidad()* int
  }
  class HabitacionEstandar {
    +calcular_tarifa(noches, factor) Decimal
    +servicios_incluidos() list
    +capacidad() int
  }
  class HabitacionSuite {
    +RECARGO: Decimal
    +DESCUENTO_LARGA: Decimal
    +calcular_tarifa(noches, factor) Decimal
    +servicios_incluidos() list
    +capacidad() int
  }
  class HabitacionFamiliar {
    -_camas_extra: int
    +CARGO_CAMA_EXTRA: Decimal
    +calcular_tarifa(noches, factor) Decimal
    +servicios_incluidos() list
    +capacidad() int
  }
  class FabricaHabitaciones {
    +crear(codigo_tipo, datos)$ Habitacion
  }

  class Servicio {
    <>
    -_nombre: str
    -_precio: Decimal
    +costo(noches, cantidad)* Decimal
  }
  class ServicioPorNoche {
    +costo(noches, cantidad) Decimal
  }
  class ServicioPorUnidad {
    +costo(noches, cantidad) Decimal
  }

  class Reserva {
    -_codigo: str
    -_cliente: Cliente
    -_habitaciones: list
    -_servicios: list
    -__estado: EstadoReserva
    +noches() int
    +huespedes() int
    +capacidad_total() int
    +agregar_habitacion(h) None
    +agregar_servicio(s, cant) None
    +calcular_total(factor) Decimal
    +confirmar() None
    +registrar_check_in() None
    +cancelar(motivo) Decimal
  }

  class RepositorioReservas {
    <>
    +guardar(reserva)* str
    +buscar(codigo)* Reserva
    +cancelar(codigo, motivo)* Decimal
  }
  class ReservaRepositorioPG {
    -_pool
    +guardar(reserva) str
    +buscar(codigo) Reserva
    +cancelar(codigo, motivo) Decimal
  }
  class GestorReservas {
    -_repo: RepositorioReservas
    +crear_reserva(dto) Reserva
    +cancelar_reserva(codigo, motivo) Decimal
  }
